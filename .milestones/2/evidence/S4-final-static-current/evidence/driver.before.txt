/-- Promoted body/consumer obligations (v3.1 Obligation 2): the two P01 helper
theorems whose proofs read `isMember`'s definition body. Checked for existence
as theorem info mentioning the operative definition ANYWHERE in the statement
(hypotheses and conclusion both harvested — a conclusion-only harvest would
falsely reject `view_mem_of_isMember`, whose occurrence sits in its
hypothesis). Mention establishes BINDING (right theorem, right operative)
only: mention is not use, and neither is sensitivity, which is established by
O5 (mandatory path) and OT2 (isolation). -/
def s4bPromotedTable : Array String := #["view_mem_of_isMember", "isMember_of_view_mem"]

partial def s4bHarvestConsts (e : Lean.Expr) (acc : Array Lean.Name) : Array Lean.Name :=
  match e with
  | .const n _ => if acc.contains n then acc else acc.push n
  | .app f a => s4bHarvestConsts a (s4bHarvestConsts f acc)
  | .lam _ t b _ => s4bHarvestConsts b (s4bHarvestConsts t acc)
  | .forallE _ t b _ => s4bHarvestConsts b (s4bHarvestConsts t acc)
  | .letE _ t v b _ => s4bHarvestConsts b (s4bHarvestConsts v (s4bHarvestConsts t acc))
  | .mdata _ b => s4bHarvestConsts b acc
  | .proj _ _ b => s4bHarvestConsts b acc
  | _ => acc

/-- Tracked owned modules, baked per-run from `git ls-files` by the wrapper
script (see TRACKED_LIT). Authority for provenance-based ownership and the
import-reach completeness check. This is derived live, never a frozen list:
a future owned module appears here automatically and either resolves (imported)
or fires the reach gap loudly. -/
def s4bTrackedModules : Array String := #["KelGroups","KelGroups.Event","KelGroups.Fold","KelGroups.Integration","KelGroups.Invariants","KelGroups.Mirrors","KelGroups.State","KelGroups.Tests","KelGroups.Types","KelGroups.Validate","KelGroups.Vote.Event","KelGroups.Vote.Fold","KelGroups.Vote.Invariants","KelGroups.Vote.State","KelGroups.Vote.Tests","KelGroups.Vote.Types","KelGroups.Vote.Validate","Reactivegas","Reactivegas.Composition","Reactivegas.CorpusExport","Reactivegas.CorpusGate","Reactivegas.Invariants","Reactivegas.Mirrors","Reactivegas.Predicates","Reactivegas.State","Reactivegas.Step","Reactivegas.Trace","Reactivegas.TraceTests","Reactivegas.Types"]

open Lean Elab Command Meta in
run_cmd do
  let env ← getEnv
  let mods := env.header.moduleNames
  let nonce ← IO.getEnv "S4B_MIRROR_NONCE"
  let nonce := nonce.getD "unset"
  let failedRef ← IO.mkRef false
  let fail (msg : MessageData) : CommandElabM Unit := do
    logError msg
    failedRef.set true
  -- 1. Discovery: TOTAL ConstantInfo classification over owned modules (F01).
  -- Provenance first, kind second with every arm explicit, result sort third.
  -- Predicate-bearing kinds (defn/induct/opaque) are result-sort tested;
  -- thm/axiom/ctor/rec are excluded with named reasons counted below; any
  -- unclassified kind fails closed with the classifier's own diagnostic
  -- (never a Lean error, never silent). Deleting any kind arm below (as the
  -- classifier-omit control does) routes real constants of that kind to the
  -- fail-closed arm instead of silently dropping them.
  let mut found : Array (String × String) := #[]
  let mut skipped : Array String := #[]
  let mut predDefnCount := 0
  let mut predInductCount := 0
  let mut predOpaqueCount := 0
  let mut nonpredDefn := 0
  let mut nonpredInduct := 0
  let mut nonpredOpaque := 0
  let mut nonowned := 0
  let mut exclThm := 0
  let mut exclAxiom := 0
  let mut exclCtorRec := 0
  for (n, ci) in env.constants.toList do
    let home : String :=
      match env.getModuleIdxFor? n with
      | some idx =>
          match mods[idx]? with
          | some m => toString m
          | none => "<no-mod>"
      | none => "<current-file>"
    if !s4bTrackedModules.contains home then
      nonowned := nonowned + 1
      continue
    match ci with
    | .defnInfo _ | .inductInfo _ | .opaqueInfo _ =>
        let kindTag : String :=
          match ci with
          | .defnInfo _ => "defn"
          | .inductInfo _ => "induct"
          | .opaqueInfo _ => "opaque"
          | _ => "UNREACHABLE-INNER"
        if kindTag == "UNREACHABLE-INNER" then do
          fail m!"MIRROR-UNCLASSIFIED-KIND {home} :: {toString n} (fail-closed: inner classification unreachable)"
          continue
        let verdict : Option Bool ← liftTermElabM (try
          forallTelescopeReducing ci.type fun _ cod => do
            let c ← whnf cod
            match c with
            | .sort .zero => pure (some true)
            | .sort _ => pure (some false)
            | _ => pure (some false)
          catch _ => pure none)
        match verdict with
        | some true =>
            found := found.push (home, toString n)
            if kindTag == "defn" then predDefnCount := predDefnCount + 1
            else if kindTag == "induct" then predInductCount := predInductCount + 1
            else predOpaqueCount := predOpaqueCount + 1
        | some false =>
            if kindTag == "defn" then nonpredDefn := nonpredDefn + 1
            else if kindTag == "induct" then nonpredInduct := nonpredInduct + 1
            else nonpredOpaque := nonpredOpaque + 1
        | none => skipped := skipped.push (toString n)
    | .thmInfo _ => exclThm := exclThm + 1
    | .axiomInfo _ => exclAxiom := exclAxiom + 1
    | .ctorInfo _ | .recInfo _ => exclCtorRec := exclCtorRec + 1
    | _ =>
        fail m!"MIRROR-UNCLASSIFIED-KIND {home} :: {toString n} (fail-closed: no classification rule covers this declaration kind)"
        continue
  if found.size == 0 then fail m!"MIRROR-EMPTY-DISCOVERY"
  for sk in skipped do logInfo m!"MIRROR-SKIP {sk}"
  logInfo m!"MIRROR-KIND-CENSUS pred(defn/induct/opaque)={predDefnCount}/{predInductCount}/{predOpaqueCount} nonpred(defn/induct/opaque)={nonpredDefn}/{nonpredInduct}/{nonpredOpaque} nonowned={nonowned} excluded(thm/axiom/ctor-rec)={exclThm}/{exclAxiom}/{exclCtorRec} sort-undecided={skipped.size}"
  -- 1b. Import-reach completeness (Boundary 1): every tracked owned module
  -- must be present in the imported closure. A tracked-but-unimported module
  -- fails closed NAMING it plus remediation (module-level refusal); predicate
  -- naming inside it follows once imported (O2 demonstrates both directions
  -- at CorpusExport). Never silent, never narrowing.
  for t in s4bTrackedModules do
    let present := mods.any (fun m => toString m == t)
    if !present then
      fail m!"MIRROR-IMPORT-REACH-GAP {t} (tracked owned module outside the checker's import closure; import it into an umbrella or the driver)"
  -- 2. Structural below-rule: `<I>.below` of a discovered induct `I` is
  -- eliminator machinery, not an owned predicate. Logged, never silent.
  let mut covered : Array String := #[]
  let mut belowCount := 0
  for (_, s) in found do
    if s.endsWith ".below" then
      let parent := s.dropRight ".below".length
      match env.find? parent.toName with
      | some (.inductInfo _) =>
          logInfo m!"MIRROR-BELOW-EXCLUDED {s} (machinery of discovered induct {parent})"
          covered := covered.push s
          belowCount := belowCount + 1
      | _ => pure ()
  -- 3. Per-identity reconciliation of the 19 claimed rows.
  let mut rowCount := 0
  for (prop, mirrorOpt, thm) in s4bCorrTable do
    if !found.any (fun (_, s) => s == prop) then
      fail m!"MIRROR-STALE-ROW {prop} (tabled but not discovered)"
      continue
    covered := covered.push prop
    match mirrorOpt with
    | some m =>
        match env.find? m.toName with
        | some _ => pure ()
        | none => fail m!"MIRROR-MISSING {prop} (no const {m})"
    | none => pure ()
    match env.find? thm.toName with
    | none => fail m!"MIRROR-THEOREM-MISSING {prop} (no const {thm})"
    | some ti =>
        match ti with
        | .thmInfo _ => pure ()
        | _ => fail m!"MIRROR-NOT-THEOREM {prop} ({thm} is not theorem info)"
        let shape ← liftTermElabM (try
          forallTelescopeReducing ti.type fun _ ty => do
            let t ← whnf ty
            match t with
            | .app (.app (.const ``Iff _) lhs) rhs =>
                -- NOTE: no whnf on `lhs` here: unfolding would dissolve
                -- def-Props into their bodies and hide the head const.
                match lhs.getAppFn with
                | .const pn _ =>
                    if toString pn == prop then
                      match mirrorOpt with
                      | some m =>
                          pure (s4bHarvestConsts rhs #[] |>.contains m.toName)
                      | none => pure true
                    else pure false
                | _ => pure false
            | _ => pure false
          catch _ => pure false)
        if !shape then fail m!"MIRROR-SHAPE-MISMATCH {prop} (theorem {thm} is not an Iff off {prop} mentioning its mirror)"
        else
          logInfo m!"MIRROR-ROW {prop} {mirrorOpt.getD "(inline-expr)"} {thm} OK"
          rowCount := rowCount + 1
  -- 3b. Promoted obligations: existence as theorems mentioning the operative def.
  let mut promotedCount := 0
  for p in s4bPromotedTable do
    match env.find? p.toName with
    | none => fail m!"MIRROR-PROMOTED-MISSING {p}"
    | some pi =>
        match pi with
        | .thmInfo _ => pure ()
        | _ => fail m!"MIRROR-PROMOTED-NOT-THEOREM {p}"
        let mentioned ← liftTermElabM (try
          forallTelescopeReducing pi.type fun xs ty => do
            let mut acc := s4bHarvestConsts ty #[]
            for x in xs do
              acc := s4bHarvestConsts (← inferType x) acc
            pure (acc.contains `KelGroups.GroupView.isMember)
          catch _ => pure false)
        if !mentioned then fail m!"MIRROR-PROMOTED-SHAPE-MISMATCH {p} (does not mention the operative definition)"
        else
          logInfo m!"MIRROR-PROMOTED {p} OK"
          promotedCount := promotedCount + 1
  -- 4. Named exceptions must be discovered (else the exception is stale).
  for e in s4bExceptions do
    if !found.any (fun (_, s) => s == e) then
      fail m!"MIRROR-STALE-EXCEPTION {e}"
    else
      covered := covered.push e
      logInfo m!"MIRROR-EXCEPTION {e} OK"
  -- 5. Anything discovered but unaccounted for is an uncovered owned predicate.
  for (home, s) in found do
    if !covered.contains s then
      fail m!"MIRROR-UNCOVERED {home} :: {s}"
  -- 6. Orphan Bool mirrors: every Bool def in the new mirror modules must be
  -- claimed by exactly one table row.
  let mut claimedMirrors : Array String := #[]
  for (_, mOpt, _) in s4bCorrTable do
    match mOpt with
    | some m => claimedMirrors := claimedMirrors.push m
    | none => pure ()
  let mut orphanCount := 0
  for (n, ci) in env.constants.toList do
    match ci with
    | .defnInfo _ => pure ()
    | _ => continue
    let home : String :=
      match env.getModuleIdxFor? n with
      | some idx =>
          match mods[idx]? with
          | some m => toString m
          | none => "<no-mod>"
      | none => "<current-file>"
    if !(home == "Reactivegas.Mirrors" || home == "KelGroups.Mirrors") then continue
    let isBool ← liftTermElabM (try
      forallTelescopeReducing ci.type fun _ cod => do
        let c ← whnf cod
        match c with
        | .const ``Bool _ => pure true
        | _ => pure false
      catch _ => pure false)
    if isBool then
      let s := toString n
      if !claimedMirrors.contains s then
        fail m!"MIRROR-ORPHAN {s} (Bool mirror in {home} claimed by no row)"
      else orphanCount := orphanCount + 1
  logInfo m!"MIRROR-SUMMARY rows={rowCount} exceptions={s4bExceptions.size} below={belowCount} orphans-checked={orphanCount} discovered={found.size} promoted={promotedCount}"
  logInfo m!"MIRROR-KIND-CENSUS-OK pred={predDefnCount + predInductCount + predOpaqueCount} excluded-thm={exclThm} unclassified=0"
  let failed ← failedRef.get
  if failed then fail m!"MIRROR-CHECK-FAILED"
  else do
    logInfo m!"MIRROR-CHECK-OK rows={rowCount} exceptions={s4bExceptions.size} discovered={found.size} promoted={promotedCount} tracked={s4bTrackedModules.size}"
    let names := (found.map (·.2)).qsort (· < ·)
    let receipt := "MIRROR-CHECK-OK rows=" ++ toString rowCount ++
      " exceptions=" ++ toString s4bExceptions.size ++
      " discovered=" ++ toString found.size ++
      " promoted=" ++ toString promotedCount ++
      " tracked=" ++ toString s4bTrackedModules.size ++ "\n" ++
      "nonce=" ++ nonce ++ "\n" ++
      "\n".intercalate (names.toList.map (fun s => "extent " ++ s)) ++ "\n" ++
      "\n".intercalate (s4bTrackedModules.toList.map (fun s => "tracked " ++ s)) ++ "\n"
    IO.FS.writeFile ".lake/s4b-mirror-receipt" receipt
    logInfo m!"MIRROR-RECEIPT-WROTE nonce={nonce}"
